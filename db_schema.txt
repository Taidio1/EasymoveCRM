CREATE TABLE public.clients (
  Name text,
  Status text CHECK ("Status" = ANY (ARRAY['Aktywny'::text, 'Nieaktywny'::text, 'W trakcie'::text, 'Zakończony'::text, 'Zaplanowany'::text, 'Zawieszony'::text, 'Planowany'::text])),
  CelPobytu text,
  PodLegPob text,
  Phone text,
  Adres text,
  Email text,
  Birthday text,
  Notes text,
  Creator text,
  TotalSpend text,
  StatusPla text CHECK (("StatusPla" = ANY (ARRAY['Opłacone'::text, 'W trakcie płatności'::text])) OR "StatusPla" IS NULL),
  Doc text,
  NumerSprawy text,
  Inspektor text,
  Firma text,
  FormWni boolean CHECK ("FormWni" = ANY (ARRAY[true, false])),
  ZalNrJed boolean CHECK ("ZalNrJed" = ANY (ARRAY[true, false])),
  KopiaPasz boolean CHECK ("KopiaPasz" = ANY (ARRAY[true, false])),
  ZalBlue boolean CHECK ("ZalBlue" = ANY (ARRAY[true, false])),
  CzteZdjecia boolean CHECK ("CzteZdjecia" = ANY (ARRAY[true, false])),
  Pelnomocnictwo boolean CHECK ("Pelnomocnictwo" = ANY (ARRAY[true, false])),
  id integer GENERATED ALWAYS AS IDENTITY NOT NULL UNIQUE,
  DataZloWnio timestamp with time zone,
  DataWydWni date,
  DataOdbKartyPob date,
  DataOdbDecyzji date,
  DataZakLegPob date,
  CreatedDate timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  country_id integer,
  country_name text,
  CONSTRAINT clients_pkey PRIMARY KEY (id),
  CONSTRAINT fk_clients_country FOREIGN KEY (country_id) REFERENCES public.countries(id)
);
CREATE TABLE public.countries (
  id integer NOT NULL DEFAULT nextval('countries_id_seq'::regclass),
  name character varying NOT NULL UNIQUE,
  code character varying,
  created_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT countries_pkey PRIMARY KEY (id)
);
CREATE TABLE public.profiles (
  id uuid NOT NULL,
  first_name text,
  last_name text,
  role text,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()),
  avatar_url text,
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
